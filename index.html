-- Optimizing Enterprise Module Reporting & Audit Readiness
WITH WorkflowMetrics AS (
  SELECT 
    module_code,
    COUNT(workflow_id) AS total_executions,
    AVG(duration_ms)   AS avg_query_time_ms,
    SUM(CASE WHEN audit_status = 'VERIFIED' THEN 1 ELSE 0 END) AS compliant_runs
  FROM dbo.OperationalWorkflows WITH (NOLOCK)
  WHERE execution_date >= DATEADD(MONTH, -6, GETDATE())
  GROUP BY module_code
)
SELECT 
  module_code,
  total_executions,
  avg_query_time_ms,
  CAST((compliant_runs * 100.0 / NULLIF(total_executions, 0)) AS DECIMAL(5,2)) AS compliance_rate_pct
FROM WorkflowMetrics
ORDER BY compliance_rate_pct DESC;
