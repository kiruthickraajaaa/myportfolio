-- Enterprise Reporting & Execution Plan Optimization
WITH ModuleComplianceCTE AS (
  SELECT
    m.module_name,
    COUNT(w.workflow_id) AS total_workflows,
    AVG(w.execution_ms)  AS avg_latency_ms,
    SUM(CASE WHEN w.audit_ready = 1 THEN 1 ELSE 0 END) AS verified
  FROM dbo.EnterpriseWorkflows w WITH (NOLOCK)
  INNER JOIN dbo.BusinessModules m
    ON w.module_id = m.module_id
  WHERE m.domain IN ('HAIS_InfectionControl', 'VendorPortal')
  GROUP BY m.module_name
)
SELECT
  module_name,
  total_workflows,
  CAST((verified * 100.0 / total_workflows) AS DECIMAL(5,2)) AS audit_score_pct
FROM ModuleComplianceCTE
ORDER BY audit_score_pct DESC;
